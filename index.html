function render(lista){ const tabla = document.getElementById("tabla"); tabla.innerHTML=""; lista.forEach(persona=>{ const vacCampo = Object.keys(persona).find( k => k.toLowerCase().includes("vacacion") ); const vac = vacCampo ? String(persona[vacCampo] || "").trim() : ""; const sale = String(persona.sale || "").trim(); const ingresa = String(persona.ingresa || "").trim(); const cargo = String(persona.CARGO || "").toUpperCase(); let claseFila=""; if( cargo.includes("DIRECTOR") || cargo.includes("SUB-DIRECTOR") || cargo.includes("ENCARGADO") || cargo.includes("SECRETARIO") || cargo.includes("TRANSPORTES") || cargo.includes("VIA PUBLICA") ){ claseFila="mando-row"; } if(vac !== ""){ claseFila="vacaciones-row"; } tabla.innerHTML += ` ${persona.CHAPA || ""} ${persona["PERSONAL PMT"] || ""} ${persona.CARGO || ""} ${vac ? "VACACIONES" : ""} ${sale} ${ingresa} `; }); }
